  max(created_at) AS created_at

  )
  SELECT (SELECT count(*) FROM g)::int AS total,
  COALESCE((SELECT json_agg(json_build_object(
  'product_id', product_id, 'price_min', price_min, 'price_max', price_max, 'in_stock', in_stock

  FROM (

  ) fv LEFT JOIN (
  SELECT value, count(*)::int AS count FROM (
  SELECT size AS value, product_id FROM s WHERE size IS NOT NULL AND fc AND fm AND fp
  GROUP BY size, product_id
  ) vp GROUP BY value
  ) n ON n.value = fv.value), '[]'::json) AS size,
  COALESCE((SELECT json_agg(json_build_object(
  'value', fv.value, 'label', fv.label, 'extra', fv.extra, 'rank', fv.rank, 'count', COALESCE(n.count, 0)
  ) ORDER BY fv.rank ASC NULLS LAST, fv.label ASC, fv.value COLLATE "C" ASC)

  SELECT material AS value, min(material_label) AS label, min(NULL::text) AS extra, min(NULL::int) AS rank
  FROM v WHERE material IS NOT NULL GROUP BY material
  ) fv LEFT JOIN (
  SELECT value, count(*)::int AS count FROM (
  SELECT material AS value, product_id FROM s WHERE material IS NOT NULL AND fc AND fs AND fp
  GROUP BY material, product_id
  ) vp GROUP BY value
  ) n ON n.value = fv.value), '[]'::json) AS material;
