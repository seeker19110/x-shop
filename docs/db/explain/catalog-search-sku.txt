  FROM catalog_variant_index WHERE sales_channel_id = ANY(ARRAY(SELECT jsonb_array_elements_text($1::jsonb)))
  GROUP BY color, color_label, color_hex, color_rank, size, size_label, size_rank, material, material_label
  ), q AS (
  SELECT product_id FROM catalog_variant_index WHERE sales_channel_id = ANY(ARRAY(SELECT jsonb_array_elements_text($2::jsonb))) AND product_id IN (SELECT product_id FROM catalog_variant_index WHERE sales_channel_id = ANY(ARRAY(SELECT jsonb_array_elements_text($3::jsonb)))

  ), s AS MATERIALIZED (
  SELECT product_id, price, in_stock, created_at, color, size, material, TRUE AS fc, TRUE AS fs, TRUE AS fm, TRUE AS fp
  FROM catalog_variant_index WHERE sales_channel_id = ANY(ARRAY(SELECT jsonb_array_elements_text($8::jsonb))) AND product_id IN (SELECT product_id FROM q)
  ), g AS MATERIALIZED (
  SELECT product_id, min(price) AS price_min, max(price) AS price_max, bool_or(in_stock) AS in_stock,
  max(created_at) AS created_at
  FROM s WHERE fc AND fs AND fm AND fp GROUP BY product_id
  )
  SELECT (SELECT count(*) FROM g)::int AS total,
  COALESCE((SELECT json_agg(json_build_object(
  'product_id', product_id, 'price_min', price_min, 'price_max', price_max, 'in_stock', in_stock
  ) ORDER BY created_at DESC, product_id COLLATE "C" ASC) FROM (
  SELECT * FROM g ORDER BY created_at DESC, product_id COLLATE "C" ASC LIMIT $9
  OFFSET $10
  ) p), '[]'::json) AS items,
  COALESCE((SELECT json_agg(json_build_object(

  FROM (
  SELECT color AS value, min(color_label) AS label, min(color_hex) AS extra, min(color_rank) AS rank
  FROM v WHERE color IS NOT NULL GROUP BY color
  ) fv LEFT JOIN (
  SELECT value, count(*)::int AS count FROM (
  SELECT color AS value, product_id FROM s WHERE color IS NOT NULL AND fs AND fm AND fp
  GROUP BY color, product_id
  ) vp GROUP BY value
  ) n ON n.value = fv.value), '[]'::json) AS color,
  COALESCE((SELECT json_agg(json_build_object(
  'value', fv.value, 'label', fv.label, 'extra', fv.extra, 'rank', fv.rank, 'count', COALESCE(n.count, 0)

  FROM (
  SELECT size AS value, min(size_label) AS label, min(NULL::text) AS extra, min(size_rank) AS rank
  FROM v WHERE size IS NOT NULL GROUP BY size
  ) fv LEFT JOIN (
  SELECT value, count(*)::int AS count FROM (
  SELECT size AS value, product_id FROM s WHERE size IS NOT NULL AND fc AND fm AND fp
  GROUP BY size, product_id
  ) vp GROUP BY value
  ) n ON n.value = fv.value), '[]'::json) AS size,
  COALESCE((SELECT json_agg(json_build_object(
  'value', fv.value, 'label', fv.label, 'extra', fv.extra, 'rank', fv.rank, 'count', COALESCE(n.count, 0)
  ) ORDER BY fv.rank ASC NULLS LAST, fv.label ASC, fv.value COLLATE "C" ASC)
  FROM (
  SELECT material AS value, min(material_label) AS label, min(NULL::text) AS extra, min(NULL::int) AS rank
  FROM v WHERE material IS NOT NULL GROUP BY material
  ) fv LEFT JOIN (
  SELECT value, count(*)::int AS count FROM (
  SELECT material AS value, product_id FROM s WHERE material IS NOT NULL AND fc AND fs AND fp
  GROUP BY material, product_id
  ) vp GROUP BY value
  ) n ON n.value = fv.value), '[]'::json) AS material;
